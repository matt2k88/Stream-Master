import React, { useState, useEffect, useCallback } from 'react';
import { motion } from 'framer-motion';
import { Link } from 'react-router-dom';
import { supabase } from '@/lib/customSupabaseClient';
import { useToast } from '@/components/ui/use-toast';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardHeader, CardTitle, CardFooter } from '@/components/ui/card';
import { Loader2, Trophy, History, FileText } from 'lucide-react';
import CompetitionCard from './CompetitionCard';

const UltraFourView = ({ onHistoryClick }) => {
  const { toast } = useToast();
  const [competitions, setCompetitions] = useState([]);
  const [loading, setLoading] = useState(true);

  const fetchCompetitionData = useCallback(async () => {
    setLoading(true);
    try {
      const { data: activeComps, error: compError } = await supabase
        .from('ultra_four_competitions')
        .select('*')
        .eq('status', 'active')
        .order('closing_time', { ascending: true });

      if (compError) throw compError;
      
      setCompetitions(activeComps || []);

    } catch (error) {
      toast({ title: 'Error fetching competition data', description: error.message, variant: 'destructive' });
    } finally {
      setLoading(false);
    }
  }, [toast]);

  useEffect(() => {
    fetchCompetitionData();
  }, [fetchCompetitionData]);


  if (loading) {
    return <div className="flex justify-center items-center p-8"><Loader2 className="h-8 w-8 animate-spin" /></div>;
  }

  return (
    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
      {competitions.length > 0 ? (
        competitions.map(comp => <CompetitionCard key={comp.id} competition={comp} />)
      ) : (
        <Card className="glass-effect text-center">
          <CardHeader>
            <Trophy className="mx-auto h-12 w-12 text-orange-400" />
            <CardTitle>No Active Competition</CardTitle>
          </CardHeader>
          <CardContent>
            <p className="text-muted-foreground">There isn't an Ultra Four competition running at the moment. Check back soon!</p>
          </CardContent>
        </Card>
      )}
      <Card className="glass-effect">
        <CardFooter className="flex-col sm:flex-row justify-center items-center gap-4 pt-6">
            <Button variant="outline" onClick={onHistoryClick}>
              <History className="mr-2 h-4 w-4" /> View Competition History
            </Button>
            <Button variant="outline" asChild>
              <Link to="/ultra-four/terms">
                <FileText className="mr-2 h-4 w-4" /> Terms & Conditions
              </Link>
            </Button>
        </CardFooter>
      </Card>
    </motion.div>
  );
};

export default UltraFourView;
